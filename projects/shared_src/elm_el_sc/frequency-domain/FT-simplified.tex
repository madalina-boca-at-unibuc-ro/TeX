\chapter{The Fourier transform of the emitted radiation}

\section{The Fourier transform of the Faraday tensor}

We want to calculate the Fourier transform of the Faraday tensor
\begin{align}
    F(\omega, {\bf x}) = \frac1{2\pi}\int\limits_{-\infty}^{\infty} dt e^{i\omega t}F(t,{\bf x})
\end{align}
We start from the simplified Jackson's formula (\ref{JacksonF})
\begin{align}
    F^{\alpha\beta}(x) =\left.\frac{e}{4\pi\epsilon_0c}\frac1{u\cdot R_0} \frac{d}{d\tau}\left[ \frac{R_0^{\alpha}u^{\beta}-R_0^{\beta}u^{\alpha}}{R_0\cdot u}\right]\right|_{\tau=\tau_r}.
\end{align}

So the Fourier transform of the Faraday tensor becomes
\begin{align}
    F(\omega, {\bf x}) = \frac1{2\pi}\frac{e}{4\pi\epsilon_0c}\int\limits_{-\infty}^{\infty} dt e^{i\omega t}\left\{\left.\frac1{u\cdot R_0} \frac{d}{d\tau}\left[ \frac{R_0^{\alpha}u^{\beta}-R_0^{\beta}u^{\alpha}}{R_0\cdot u}\right]\right\}\right|_{\tau=\tau_r}.
\end{align}
The relation between the laboratory time $t$ and the retarded proper time $\tau_r$ is given in (\ref{ct-taur-rel})
\begin{align}
    ct= r^0(\tau_r)+|{\bf x}-{\bf r}(\tau_r)| \equiv r_0^0(\tau_r)+|{\bf x}_0-{\bf r}_0(\tau_r)|.
\end{align}
We change tha variable of integration for $t$ to $\tau_r$ in the previous integral, using
\begin{align}
    \frac{d{ct}}{d\tau_r} = \frac{dr_0^0(\tau_r)}{d\tau_r} + \frac{d|{\bf x}_0 -{\bf r}_0(\tau_r)|}{d\tau_r} = u^0(\tau_r) - {\bf u}(\tau_t)\cdot\frac{{\bf x}_0 -{\bf r}_0(\tau_r)}{{|\bf x}_0 -{\bf r}_0(\tau_r)|}
\end{align}
or, using the unity four vectors $R_0$ and   $n_{R_0}$ defined in (\ref{defnr})
\begin{align}
    n_{R_0}=(1,{\bf n}_{R_0}),\quad {\bf n}_{R_0}=\frac{{\bf x}_0-{\bf r}_0(\tau_r)}{|{\bf x}_0-{\bf r}_0(\tau_r)|},\quad  R_0 =  |{\bf R}_0|n_{R_0}
\end{align}
we get 
\begin{importantbox}\begin{align} 
    \frac{d(ct)}{d\tau_r} = {n_{R_0}(\tau_R)\cdot u(\tau_r)}
\end{align}\end{importantbox}
Using the notation 
\begin{importantbox}
\begin{align} 
    k = \frac{\omega}c\label{defk}
\end{align}    
\end{importantbox}
and denoting the variable of integration by $\tau$ we have 
\begin{align} 
F^{\alpha\beta}(c k, {\bf x}) &= \frac1{2\pi}\frac{e}{4\pi\epsilon_0c}\int\limits_{-\infty}^{\infty} d\tau e^{i k (r_0^0(\tau)+|{\bf x}_0-{\bf r}_0(\tau_r)|)}\frac{u\cdot n_{R_0}}{u\cdot R_0} \frac{d}{d\tau}\left[ \frac{R_0^{\alpha}u^{\beta}-R_0^{\beta}u^{\alpha}}{R_0\cdot u}\right]\nonumber\\
&= \frac1{2\pi}\frac{e}{4\pi\epsilon_0c}\int\limits_{-\infty}^{\infty} d\tau e^{i k (r_0^0+|{\bf R}_0|)}\frac{1}{|{\bf R}_0|} \frac{d}{d\tau}\left[ \frac{n_0^{\alpha}u^{\beta}-n_0^{\beta}u^{\alpha}}{n_0\cdot u}\right]\nonumber\\
\end{align}
We perform an integration by parts, and neglect the boundary terms, which contribute only for $\omega\rightarrow0$, and we use 
\begin{align} 
\frac{d}{d\tau}e^{i k (r_0^0+|{\bf R}_0|)}\frac1{|{\bf R}_0|} = i k (u\cdot n_{R_0}) e^{i k (r_0^0+|{\bf R}_0|)}\frac1{|{\bf R}_0|} - e^{i k (r_0^0+|{\bf R}_0|)}\frac{{\bf n}_{R_0}\cdot {\bf u}}{|{\bf R}_0|^2}
\end{align}
After the integration by parts, the result is
\begin{align}
F^{\alpha\beta}(c k, {\bf x}) &= \frac1{2\pi}\frac{e}{4\pi\epsilon_0c}\int\limits_{-\infty}^{\infty} d\tau e^{i k (r_0^0+|{\bf R}_0|)}\left\{ -ik \frac{  (u\cdot n_{R_0})}{|{\bf R}_0|}+\frac{{\bf n}_{R_0}\cdot{\bf u}}{|{\bf R}_0|^2}\right\} \frac{n_{R_0}^{\alpha}u^{\beta}-n_{R_0}^{\beta}u^{\alpha}}{n_{R_0}\cdot u}
\end{align}

\begin{highlightbox}
\begin{align}
F^{\alpha\beta}(c k,{\bf x}) &= F^{\alpha\beta}_{l}(c k,{\bf x}) + F^{\alpha\beta}_{s}(c k,{\bf x}),
\\
F^{\alpha\beta}_{l}(c k,{\bf x}) &=
\frac1{2\pi}\frac{e}{4\pi\epsilon_0c}\int\limits_{-\infty}^{\infty} d\tau\,
e^{i k (r_0^0+|{\bf R}_0|)}
\left(
-ik \frac{u\cdot n_{R_0}}{|{\bf R}_0|}
\right)
\frac{n_{R_0}^{\alpha}u^{\beta}-n_{R_0}^{\beta}u^{\alpha}}{n_{R_0}\cdot u},
\\
F^{\alpha\beta}_{s}(c k,{\bf x}) &=
\frac1{2\pi}\frac{e}{4\pi\epsilon_0c}\int\limits_{-\infty}^{\infty} d\tau\,
e^{i k (r_0^0+|{\bf R}_0|)}
\left(
\frac{{\bf n}_{R_0}\cdot{\bf u}}{|{\bf R}_0|^2}
\right)
\frac{n_{R_0}^{\alpha}u^{\beta}-n_{R_0}^{\beta}u^{\alpha}}{n_{R_0}\cdot u}.
\end{align}
\end{highlightbox}
The notations are
\begin{highlightbox}
\begin{align}
k = \frac{\omega}{c},\quad 
R_0 = (|{\bf R}_0|,{\bf R}_0)=|{\bf R}_0| (1,{\bf n}_{R_0}) = |{\bf R}_0|n_{R_0}, \qquad {\bf R}_0 = {\bf x}_0-{\bf r}_0(\tau),\quad {\bf n}_{R_0} = \frac{{\bf R}_0}{|{\bf R}_0|}.
\end{align}
\end{highlightbox}
In the integrals the integrands must be evaluated as functions of $\tau$, the variable of integration.
