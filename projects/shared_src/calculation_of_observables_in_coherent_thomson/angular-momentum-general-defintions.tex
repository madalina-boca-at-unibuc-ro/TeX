\section{General decomposition of angular momentum}

The electromagnetic angular-momentum density about a chosen origin is
\begin{align}
  {\bf l}(t,{\bf x})={\bf x}\times{\bf g}(t,{\bf x}),
  \label{eq:angular-momentum-density-general}
\end{align}
where ${\bf g}={\bf S}/c^2$ is the linear-momentum density.  Correspondingly,
the flux density of the $i$th component of angular momentum in the $l$th
direction is
\begin{align}
  M_{li}(t,{\bf x})
  =\epsilon_{ijk}x_jT_{kl}(t,{\bf x}),
  \label{eq:angular-momentum-flux-tensor-general}
\end{align}
with $T_{kl}$ defined in
Eq.~\eqref{eq:linear-momentum-flux-tensor}.  These definitions are exact.

We consider a detector parallel to the $Oxy$ plane, with normal along the
positive $x_3$ direction.  The spectral flux density of the third component
of angular momentum through this detector is therefore
\begin{align}
  \frac{d\mathcal M_{33,+}}{d\omega}
  =x_1\frac{d\mathcal T_{23,+}}{d\omega}
  -x_2\frac{d\mathcal T_{13,+}}{d\omega},
  \qquad \omega>0,
  \label{eq:spectral-total-angular-momentum-flux-density}
\end{align}
and its integral over the detector gives the total angular-momentum spectrum,
\begin{align}
  \frac{d\mathcal J^{\mathrm{tot}}_3}{d\omega}
  =\int_A dA\,
  \frac{d\mathcal M_{33,+}}{d\omega}.
  \label{eq:spectral-total-angular-momentum-flux}
\end{align}

The total angular momentum is separated first into extrinsic and intrinsic
parts,
\begin{align}
  \frac{d\mathcal J^{\mathrm{tot}}_3}{d\omega}
  =\frac{d\mathcal J^{\mathrm{ext}}_3}{d\omega}
  +\frac{d\mathcal J^{\mathrm{int}}_3}{d\omega}.
  \label{eq:extrinsic-intrinsic-angular-momentum-split}
\end{align}
\begin{importantbox}
  To define the extrinsic part, let $(X_c,Y_c)$ be the centroid of the energy
  crossing the detector at the frequency $\omega$,
  \begin{align}
    X_c(\omega)
    &=\frac{\displaystyle\int_A dA\,x_1
    \dfrac{d\mathcal S^3_+}{d\omega}}
    {\displaystyle\int_A dA\,
    \dfrac{d\mathcal S^3_+}{d\omega}}, \quad
    Y_c(\omega)
    =\frac{\displaystyle\int_A dA\,x_2
    \dfrac{d\mathcal S^3_+}{d\omega}}
    {\displaystyle\int_A dA\,
    \dfrac{d\mathcal S^3_+}{d\omega}}.
    \label{eq:spectral-energy-flux-centroid}
  \end{align}
\end{importantbox}
Recall that $\mathcal P_i$, defined in
Eq.~\eqref{eq:spectral-screen-integrated-linear-momentum}, denotes the total
$i$th component of linear momentum transported through the detector.  In
particular, the transverse components entering the extrinsic angular
momentum are
\begin{align}
  \frac{d\mathcal P_i}{d\omega}
  =\int_A dA\,\frac{d\mathcal T_{i3,+}}{d\omega},
  \qquad i=1,2,\quad \omega>0.
  \label{eq:spectral-transverse-momentum-flux}
\end{align}
\begin{importantbox}
  The extrinsic angular momentum is the angular momentum of this centroid and
  the net transverse momentum flux:
  \begin{align}
    \frac{d\mathcal J^{\mathrm{ext}}_3}{d\omega}
    =X_c\frac{d\mathcal P_2}{d\omega}
    -Y_c\frac{d\mathcal P_1}{d\omega}.
    \label{eq:spectral-extrinsic-angular-momentum-flux}
  \end{align}
  It is the contribution associated with propagation that is not directed
  along the detector normal.  In particular, it vanishes when the net
  transverse momentum fluxes vanish, or if the Poynting vector is distributed symmetric about the origin.  It depends on the choice of transverse
  origin.
\end{importantbox}
The intrinsic contribution is the remainder evaluated about the energy-flux
centroid,
\begin{align}
  \frac{d\mathcal J^{\mathrm{int}}_3}{d\omega}
  &=\frac{d\mathcal J^{\mathrm{tot}}_3}{d\omega}
  -\frac{d\mathcal J^{\mathrm{ext}}_3}{d\omega}=\int_A dA\left[
    (x_1-X_c)\frac{d\mathcal T_{23,+}}{d\omega}
    -(x_2-Y_c)\frac{d\mathcal T_{13,+}}{d\omega}
  \right].
  \label{eq:spectral-intrinsic-angular-momentum-flux}
\end{align}
\begin{importantbox}
  If $X_c=Y_c=0$ then
  \begin{align}
    \frac{d\mathcal J^{\mathrm{int}}_3}{d\omega}
    &=\frac{d\mathcal J^{\mathrm{tot}}_3}{d\omega}
    -\frac{d\mathcal J^{\mathrm{ext}}_3}{d\omega}=\int_A dA\left[
      x_1\frac{d\mathcal T_{23,+}}{d\omega}
      -x_2\frac{d\mathcal T_{13,+}}{d\omega}
    \right].
    \label{eq:spectral-intrinsic-angular-momentum-flux-symmetric}
  \end{align}
\end{importantbox}

Unlike the extrinsic part, this quantity is invariant under a translation of
the transverse coordinate origin accompanied by the same translation of the
centroid.

Finally, the intrinsic angular momentum is separated into spin and intrinsic
orbital contributions,
\begin{importantbox}
  \begin{align}
    \frac{d\mathcal J^{\mathrm{int}}_3}{d\omega}
    =\frac{d\mathcal J^{\mathrm{spin}}_3}{d\omega}
    +\frac{d\mathcal J^{\mathrm{orb,int}}_3}{d\omega}.
    \label{eq:spin-orbital-angular-momentum-split}
  \end{align}
  The spin term describes the polarization contribution, whereas the intrinsic
  orbital term describes the spatial structure of the field about its
  energy-flux centroid.
\end{importantbox}
