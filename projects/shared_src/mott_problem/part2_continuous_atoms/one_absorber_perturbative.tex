\section{the one absorber system in the perturbative regime}\label{section-perturbative-one-particle}

we consider the case of only one atom, located at $x=a_1$. we can write the initial state as
\begin{align}
  |\psi_i\rangle = \int\limits_{-\infty}^{\infty}dP C(P)|P\rangle\otimes|0(a_1)\rangle
\end{align}
where $|0(a_1)\rangle$ is the ground state of the harmonic oscillator with the center at $x=a_1$.
the set of final states has the form
\begin{align}
  |\psi_f\rangle = |P_f,n_f\rangle = |P_f\rangle\otimes|n_f(a_1)\rangle
\end{align}

the matrix element of the interaction potential in the basis of the energy eigenstates is
\begin{align}
  \langle n_1(a_1), P_1|H_I|n_2(a_1),P_2\rangle & = \frac{V_0}{2\pi\hbar}\int \limits_{-\infty}^{\infty} dx\int \limits_{-\infty}^{\infty} dX e^{-\frac{i}{\hbar}X(P_1-P_2)}u_{n_1}(x-a_1)u_{n_2}(x-a_1)e^{-\frac{(x-X)^2}{2\sigma_V^2}}\nonumber                         \\
                                                & =\frac{V_0}{2\pi\hbar}\int \limits_{-\infty}^{\infty} dx\int \limits_{-\infty}^{\infty} dX e^{-\frac{i}{\hbar}(X-x)(P_1-P_2)}e^{-\frac{i}{\hbar}x(P_1-P_2)}u_{n_1}(x-a_1)u_{n_2}(x-a_1)e^{-\frac{(x-X)^2}{2\sigma_V^2}}
\end{align}
after the change of variable $X-x=y$ the previous matrix element becomes
\begin{align}
  \langle n_1(a_1), P_1|H_I|n_2(a_1),P_2\rangle & = \frac{V_0}{2\pi\hbar}\int \limits_{-\infty}^{\infty} dy e^{-\frac{i}{\hbar}y(P_1-P_2)}e^{-\frac{y^2}{2\sigma_V^2}}\int \limits_{-\infty}^{\infty} dx e^{-\frac{i}{\hbar}x(P_1-P_2)}u_{n_1}(x-a_1)u_{n_2}(x-a_1)
\end{align}
The integral over $y$ can be calculated using (\ref{usefulid2}), with $\lambda = \frac1{2\sigma_V^2}$ and $k = \frac{P_1-P_2}{\hbar}$
\begin{align}
  \int \limits_{-\infty}^{\infty} dy e^{-\frac{i}{\hbar}y(P_1-P_2)}e^{-\frac{y^2}{2\sigma_V^2}} = \sqrt{{2\pi\sigma_V^2}} e^{-\frac{(P_1-P_2)^2\sigma_V^2}{2\hbar^2}}
\end{align}
The integral over $x$ is calculated after a further change of variable $x-a_1=z$
\begin{align}
  \int \limits_{-\infty}^{\infty} dx e^{-\frac{i}{\hbar}x(P_1-P_2)}u_{n_1}(x-a_1)u_{n_2}(x-a_1) = e^{-\frac{i}{\hbar}a_1(P_1-P_2)}\int \limits_{-\infty}^{\infty} dz e^{-\frac{i}{\hbar}z(P_1-P_2)}u_{n_1}(z)u_{n_2}(z)
\end{align}
The remaining integral can be computed using (\ref{me-ho}), with $q = \frac{P_1-P_2}{\hbar}$ and $\alpha = \sqrt{\frac{m\omega}{\hbar}}$
\begin{align}
  \int \limits_{-\infty}^{\infty} dz e^{-\frac{i}{\hbar}z(P_1-P_2)}u_{n_1}(z)u_{n_2}(z) = \sqrt{\frac{n_<!}{n_>!}} \left(-\frac{i (P_1-P_2)  }{\sqrt{2}\hbar\alpha}\right)^{n_> - n_<} e^{-\frac{(P_1-P_2)^2 }{4\hbar^2\alpha^2}} \, L_{n_<}^{(n_> - n_<)}\!\left(\frac{(P_1-P_2)^2 }{2\hbar^2\alpha^2}\right)
\end{align}

The transition amplitude to calculate is (\ref{first_order_transition_amplitude})
\begin{align}
  {\cal A}_{fi}^{(1)} & =e^{-\frac{i}{\hbar}E_{\gamma_f}(t_f-t_i)}\sum_\gamma \frac{c_\gamma^{(i)}}{{E_{\gamma_f}- E_\gamma}} \langle \gamma_f|H_I|\gamma\rangle  \left(1-e^{\frac{i}{\hbar}({E_{\gamma_f}- E_\gamma})(t_f-t_i)} \right)
\end{align}
which in our case is
\begin{align}
   & \gamma_f = {P_f, n_f},\qquad \gamma = {P, 0},\qquad c_\gamma^{(i)} = C(P)\nonumber                                                                                                                                                                                                                                                                                                                       \\
   & E_{\gamma_f}- E_\gamma = \hbar\omega_{(P_f,n_f) (P,0)} = {n_f\hbar\omega+ \frac{P_f^2-P^2}{2m}}                                                                                                                                                                                                                                                                                                          \\
   & \langle \gamma_f|H_I|\gamma\rangle = \langle P_f,n_f |H_I|P,0\rangle = \frac{V_0}{2\pi\hbar}\sqrt{2\pi \sigma_V^2 }e^{-\frac{\left(P_f-P\right)^2 \sigma_V^2 }{2\hbar ^2}}e^{-\frac{i}{\hbar}a_1(P_f-P)}{\frac{1}{\sqrt{n_f!}}}\left(-\frac{i (P_f-P) }{\sqrt{2}\hbar\alpha }\right)^{n_f} e^{-\frac{(P_f-P)^2 }{4\hbar^2\alpha ^2}} \, L_{0}^{(n_f)}\!\left(\frac{(P_f-P)^2 }{2\hbar^2\alpha ^2}\right)
\end{align}
since the Laguerre polynomial reduces to 1, the  result is
\begin{align}
  \langle P_f,n_f |H_I|P,0\rangle = \frac{V_0\sigma_V}{\sqrt{2\pi}\hbar {\sqrt{n_f!}}}e^{-\frac{\left(P_f-P\right)^2 }{2\hbar ^2}(\sigma_V^2 +
      \frac{1}{2\alpha^2})}e^{-\frac{i}{\hbar}a_1(P_f-P)}\left(-\frac{i (P_f-P)  }{\sqrt{2}\hbar\alpha}\right)^{n_f}\label{matrix_element_one_atom}
\end{align}

Finally, the transition amplitude is given by (up to an irrelevant factor of $e^{-\frac{i}{\hbar}E_{\gamma_f}(t_f-t_i)}$)
\begin{align}
  \label{A1_one_atom_2}    {\cal A}_{fi}^{(1)} & =\int\limits_{-\infty}^{\infty}dP C(P) \langle P_f,n_f |H_I|P,0\rangle  \frac{1-e^{\frac{i}{\hbar}(n_f\hbar\omega+\frac{P_f^2-P^2}{2m})(t_f-t_i)}}{n_f\hbar\omega+\frac{P_f^2-P^2}{2m}}
\end{align}
where $C(P)$ is given in eq. (\ref{C(P)-projectile}) and $\langle P_f,n_f |H_I|P,0\rangle$ is given in eq. (\ref{matrix_element_one_atom}).
