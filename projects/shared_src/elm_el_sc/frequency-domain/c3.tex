\chapter{The Fourier series of the emitted field in Thomson scattering - the case of the circular motion}


In this chapter we will consider the trajectory
\begin{align}
    r(t)=a\left({\bm e}_x\cos\omega_L t+{\bm e}_y\sin\omega_L t\right), \qquad {\boldsymbol{\beta}}(t)=\frac{\omega_L}ca\left(-{\bm e}_x\sin\omega_Lt+{\bm e}_y\cos\omega_Lt\right)
\end{align}
with these
\begin{align}
     & {\bm r}^2_0(t)=a^2                                                                                                                                       \\
     & {\bm n}_0\cdot{\bm r}_0(t)=a\sin\theta_0(\cos\phi_0\cos\omega_Lt+\sin\phi_0\sin\omega_Lt)=a\sin\theta_0\cos(\omega_Lt-\phi_0)                            \\
     & {\bm n}_0\times{\bm r}_0(t)=a(-{\bm e}_x\cos\theta_0\sin(\omega_L t)+{\bm e}_y \cos\theta_0\cos(\omega_L t)-{\bm e}_z\sin\theta_0\cos(\omega_Lt-\phi_0)) \\
     & |{\bm n}_0\times{\bm r}_0(t)|^2=a^2\left(\cos^2\theta_0+\sin^2\theta_0\sin^2(\omega_Lt -\phi_0)\right)
\end{align}
\section{The Fourier series of the potentials}



\subsection{First order expansion}

\subsubsection{The scalar potential}
\begin{align}
    \Phi({\bm x}_0,\omega)=\frac{e_0}{4\pi\epsilon_0}\frac1{2\pi}\frac{e^{ik x_0}}{x_0}\int\limits_{-\infty}^{\infty}dt e^{i\omega t-i\frac{\omega}ca\sin\theta_0\cos(\omega_Lt-\phi_0)}
\end{align}
We use the Anger identity
\begin{align}
    e^{i z \cos \theta_0} \equiv \sum_{n=-\infty}^{\infty} i^n J_n(z) e^{i n \theta_0}
\end{align}
to write
\begin{align}
    e^{-ia\sin\theta_0\cos(\omega_Lt-\phi_0)}=\sum_{n=-\infty}^{\infty} i^n J_n(-\frac{\omega a}c\sin\theta_0) e^{i n (\omega_Lt-\phi_0)}
\end{align}
Using
\begin{align}
    J_n(-z)=J_{-n}(z)
\end{align}
we write the previous relation as
\begin{align}
    e^{-ia\sin\theta_0\cos(\omega_Lt-\phi_0)}=\sum_{n=-\infty}^{\infty} (-i)^n J_n(\frac{\omega a}c\sin\theta_0) e^{-i n (\omega_Lt-\phi_0)}
\end{align}
and the scalar potential is
\begin{align}
    \Phi({\bm x}_0,\omega) & =\frac{e_0}{4\pi\epsilon_0}\sum\limits_{n=-\infty}^{\infty}\frac{e^{ik x_0}}{x_0}\frac1{2\pi}\int\limits_{-\infty}^{\infty}dt e^{i\omega t} (-i)^n J_n(\frac{\omega a}c\sin\theta_0) e^{-i n (\omega_Lt-\phi_0)}\nonumber            \\
                           & =\frac{e_0}{4\pi\epsilon_0}\frac{e^{ik x_0}}{x_0}\sum\limits_{n=-\infty}^{\infty}(-i)^ne^{in\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\frac1{2\pi}\int\limits_{-\infty}^{\infty}dt e^{i\omega t}   e^{-i n \omega_Lt}\nonumber         \\
                           & =\frac{e_0}{4\pi\epsilon_0}\frac{e^{ik x_0}}{x_0}\sum\limits_{n=-\infty}^{\infty}(-i)^ne^{in\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\delta(\omega-n\omega_L)=\sum\limits_{n=-\infty}^{\infty}\Phi_n({\bm x})\delta(\omega-n\omega_L)
\end{align}
with
\begin{align}
    \Phi_n({\bm x})=\frac{e_0}{4\pi\epsilon_0}\frac{e^{ik_n x_0}}{x_0}(-i)^nJ_n(k_na\sin\theta_0)e^{in\phi_0},\qquad k_n=\frac{n\omega_L}c
\end{align}

\subsubsection{The vector potential}
\begin{align}
    {\bm A}({\bm x},\omega)= \frac{e_0}{4\pi\epsilon_0c}\frac1{2\pi}\frac{e^{ik x_0}}{x_0}\int\limits_{-\infty}^{\infty}dt {\boldsymbol{\beta}}(t)e^{i\omega t-i\frac{\omega}ca\sin\theta_0\cos(\omega_Lt-\phi_0)}
\end{align}
We write ${\boldsymbol{\beta}}(t)$ is spherical coordinates
\begin{align}
    {\boldsymbol{\beta}}(t)=\frac{\omega_L}c a(-{\bm e}_x\sin(\omega_L t)+{\bm e}_y\cos(\omega_Lt))
\end{align}
we use
\begin{align}
    {\bm e}_x=\sin\theta_0\cos\phi_0{\bm e}_r+\cos\theta_0\cos\phi_0{\bm e}_\theta-\sin\phi{\bm e}_\phi \\
    {\bm e}_y=\sin\theta_0\sin\phi_0{\bm e}_r+\cos\theta_0\sin\phi_0{\bm e}_\theta+\cos\phi{\bm e}_\phi
\end{align}
In spherical coordinates
\begin{align}
    {\boldsymbol{\beta}}(t) = & \frac{\omega_L}c a\left[-(\sin\theta_0\cos\phi_0{\bm e}_r+\cos\theta_0\cos\phi_0{\bm e}_\theta-\sin\phi{\bm e}_\phi )\sin(\omega_L t)\right.\nonumber \\
                              & \quad\quad + \left.(\sin\theta_0\sin\phi_0{\bm e}_r+\cos\theta_0\sin\phi_0{\bm e}_\theta+\cos\phi{\bm e}_\phi)\cos(\omega_Lt)\right]
\end{align}
The spherical components of $\boldsymbol{\beta}$ are
\begin{align}
     & \beta_r(t)=-\frac{\omega_L}ca\sin\theta_0(\sin\omega_Lt\cos\phi_0-\cos\omega_Lt\sin\phi_0)=-\frac{\omega_L}{c}a\sin\theta_0\sin(\omega_Lt-\phi_0)      \\
     & \beta_\theta(t)=-\frac{\omega_L}ca\cos\theta_0(\sin\omega_Lt\cos\phi_0-\cos\omega_Lt\sin\phi_0)=-\frac{\omega_L}{c}a\cos\theta_0\sin(\omega_Lt-\phi_0) \\
     & \beta_\phi(t)=\frac{\omega_L}{c}a(\cos\omega_Lt\cos\phi_0+\sin\omega_Lt\sin\phi_0)=\frac{\omega_L}ca\cos(\omega_Lt-\phi_0)
\end{align}
The spherical components of ${\bm A}$ are
\begin{align}
    A_r({\bm x},\omega) & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(-\frac{\omega_L}ca\sin\theta_0\right)\sum\limits_{n=-\infty}^{\infty}(-i)^ne^{in\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\frac1{2\pi}\int\limits_{-\infty}^{\infty}dt e^{i\omega t}   e^{-i n \omega_Lt}\sin(\omega_Lt-\phi_0)\nonumber                                             \\
                        & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(-\frac{\omega_L}ca\sin\theta_0\right)\sum\limits_{n=-\infty}^{\infty}(-i)^ne^{in\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\frac1{4\pi i}\int\limits_{-\infty}^{\infty}dt e^{i\omega t}  \left( e^{-i (n-1) \omega_Lt}e^{-i\phi_0}- e^{-i (n+1) \omega_Lt}e^{i\phi_0}\right)\nonumber \\
                        & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(-\frac{\omega_L}ca\sin\theta_0\right)\frac12\sum\limits_{n=-\infty}^{\infty}(-i)^{n+1}e^{in\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\left(\delta(\omega-(n-1)\omega_L)e^{-i\phi_0}-\delta(\omega-(n+1)\omega_L)e^{i\phi_0}\right)\nonumber                                          \\
                        & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(-\frac{\omega_L}ca\sin\theta_0\right)\frac12\sum\limits_{n=-\infty}^{\infty}(-i)^{n+1}e^{i(n-1)\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\delta(\omega-(n-1)\omega_L)\nonumber                                                                                                       \\
                        & \quad-                        \frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(-\frac{\omega_L}ca\sin\theta_0\right)\frac12\sum\limits_{n=-\infty}^{\infty}(-i)^{n+1}e^{i(n+1)\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\delta(\omega-(n+1)\omega_L)\nonumber                                                                          \\
                        & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(\frac{\omega_L}ca\sin\theta_0\right)\frac12\sum\limits_{n=-\infty}^{\infty}(-i)^{n-1}e^{i(n-1)\phi_0}J_n(\frac{(n-1)\omega_L a}c\sin\theta_0)\delta(\omega-(n-1)\omega_L)\nonumber                                                                                                 \\
                        & \quad+                        \frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(\frac{\omega_L}ca\sin\theta_0\right)\frac12\sum\limits_{n=-\infty}^{\infty}(-i)^{n+1}e^{i(n+1)\phi_0}J_n(\frac{(n+1)\omega_L a}c\sin\theta_0)\delta(\omega-(n+1)\omega_L)
\end{align}
In the previous sums we make a change of summation index $(n-1)\rightarrow n$ in the first line and $(n+1)\rightarrow n$ in the second line.
\begin{align}
    A_r({\bm x},\omega) & =\frac{e_0}{4\pi\epsilon_0c}\frac12\left(\frac{\omega_L}ca\sin\theta_0\right)\sum\limits_{n=-\infty}^{\infty}\frac{e^{ik_n x_0}}{x_0}(-i)^{n}e^{in\phi_0}(J_{n+1}(k_na\sin\theta_0)+J_{n-1}(k_na\sin\theta_0))\delta(\omega-n\omega_L)
\end{align}
We use the identity
\begin{align}
    J_{n+1}(x)+J_{n-1}(x)=\frac{2 n}{x} J_n(x)
\end{align}
which in our case becomes
\begin{align}
    J_{n+1}(k_na\sin\theta_0)+J_{n-1}(k_na\sin\theta_0)=\frac{2nc}{n\omega_L a\sin\theta_0}J_n(k_na\sin\theta_0)
\end{align}
and we have
\begin{align}
    A_r({\bm x},\omega)=\sum\limits_{n=-\infty}^{\infty}A_{rn}({\bm x})\delta(\omega-n\omega_L)
\end{align}
with
\begin{align}
    A_{rn}({\bm x})=    \frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik_n x_0}}{x_0}(-i)^ne^{in\phi_0}J_n(k_na\sin\theta_0)
\end{align}

\begin{align}
    A_\theta({\bm x},\omega) & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(-\frac{\omega_L}ca\cos\theta_0\right)\sum\limits_{n=-\infty}^{\infty}(-i)^ne^{in\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\frac1{2\pi}\int\limits_{-\infty}^{\infty}dt e^{i\omega t}   e^{-i n \omega_Lt}\sin(\omega_Lt-\phi_0)\nonumber                                             \\
                             & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(-\frac{\omega_L}ca\cos\theta_0\right)\sum\limits_{n=-\infty}^{\infty}(-i)^ne^{in\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\frac1{4\pi i}\int\limits_{-\infty}^{\infty}dt e^{i\omega t}  \left( e^{-i (n-1) \omega_Lt}e^{-i\phi_0}- e^{-i (n+1) \omega_Lt}e^{i\phi_0}\right)\nonumber \\
                             & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(-\frac{\omega_L}ca\cos\theta_0\right)\frac12\sum\limits_{n=-\infty}^{\infty}(-i)^{n+1}e^{in\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\left(\delta(\omega-(n-1)\omega_L)e^{-i\phi_0}-\delta(\omega-(n+1)\omega_L)e^{i\phi_0}\right)\nonumber                                          \\
                             & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(-\frac{\omega_L}ca\cos\theta_0\right)\frac12\sum\limits_{n=-\infty}^{\infty}(-i)^{n+1}e^{i(n-1)\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\delta(\omega-(n-1)\omega_L)\nonumber                                                                                                       \\
                             & \quad-                        \frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(-\frac{\omega_L}ca\cos\theta_0\right)\frac12\sum\limits_{n=-\infty}^{\infty}(-i)^{n+1}e^{i(n+1)\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\delta(\omega-(n+1)\omega_L)\nonumber                                                                          \\
                             & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(\frac{\omega_L}ca\cos\theta_0\right)\frac12\sum\limits_{n=-\infty}^{\infty}(-i)^{n-1}e^{i(n-1)\phi_0}J_n(\frac{(n-1)\omega_L a}c\sin\theta_0)\delta(\omega-(n-1)\omega_L)\nonumber                                                                                                 \\
                             & \quad+                        \frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(\frac{\omega_L}ca\cos\theta_0\right)\frac12\sum\limits_{n=-\infty}^{\infty}(-i)^{n+1}e^{i(n+1)\phi_0}J_n(\frac{(n+1)\omega_L a}c\sin\theta_0)\delta(\omega-(n+1)\omega_L)
\end{align}
After changing the summation indices
\begin{align}
    A_\theta({\bm x},\omega) & =\frac{e_0}{4\pi\epsilon_0c}\frac12\left(\frac{\omega_L}ca\cos\theta_0\right)\sum\limits_{n=-\infty}^{\infty}\frac{e^{ik_n x_0}}{x_0}(-i)^{n}e^{in\phi_0}(J_{n+1}(k_na\sin\theta_0)+J_{n-1}(k_na\sin\theta_0))\delta(\omega-n\omega_L)
\end{align}
with the result
\begin{align}
    A_\theta({\bm x},\omega)=\sum\limits_{n=-\infty}^{\infty}A_{\theta n}({\bm x})\delta(\omega-n\omega_L)
\end{align}
with
\begin{align}
    A_{\theta n}({\bm x})=    \frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik_n x_0}}{x_0}\cot\theta_0(-i)^ne^{in\phi_0}J_n(k_na\sin\theta_0)
\end{align}

\begin{align}
    A_\phi({\bm x},\omega) & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(\frac{\omega_L}ca\right)\sum\limits_{n=-\infty}^{\infty}(-i)^ne^{in\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\frac1{2\pi}\int\limits_{-\infty}^{\infty}dt e^{i\omega t}   e^{-i n \omega_Lt}\cos(\omega_Lt-\phi_0)\nonumber                                            \\
                           & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(\frac{\omega_L}ca\right)\sum\limits_{n=-\infty}^{\infty}(-i)^ne^{in\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\frac1{4\pi }\int\limits_{-\infty}^{\infty}dt e^{i\omega t}  \left( e^{-i (n-1) \omega_Lt}e^{-i\phi_0}+ e^{-i (n+1) \omega_Lt}e^{i\phi_0}\right)\nonumber \\
                           & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(\frac{\omega_L}ca\right)\frac12\sum\limits_{n=-\infty}^{\infty}(-i)^{n}e^{in\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\left(\delta(\omega-(n-1)\omega_L)e^{-i\phi_0}+\delta(\omega-(n+1)\omega_L)e^{i\phi_0}\right)\nonumber                                           \\
                           & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(\frac{\omega_L}ca\right)\frac12\sum\limits_{n=-\infty}^{\infty}(-i)^{n}e^{i(n-1)\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\delta(\omega-(n-1)\omega_L)\nonumber                                                                                                        \\
                           & \quad+                        \frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(\frac{\omega_L}ca\right)\frac12\sum\limits_{n=-\infty}^{\infty}(-i)^{n}e^{i(n+1)\phi_0}J_n(\frac{\omega a}c\sin\theta_0)\delta(\omega-(n+1)\omega_L)
\end{align}
After the change of summation indices
\begin{align}
    A_\phi({\bm x},\omega) & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(\frac{\omega_L}ca\right)\frac12\sum\limits_{n=-\infty}^{\infty}e^{in\phi_0}\left((-i)^{n+1}J_{n+1}(\frac{\omega a}c\sin\theta_0)+(-i)^{n-1}J_{n-1}(\frac{\omega a}c\sin\theta_0)\right)\delta(\omega-n\omega_L)\nonumber \\
                           & =\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik x_0}}{x_0}\left(\frac{\omega_L}ca\right)(-i)\frac12\sum\limits_{n=-\infty}^{\infty}e^{in\phi_0}(-i)^n\left( J_{n+1}(\frac{\omega a}c\sin\theta_0)-J_{n-1}(\frac{\omega a}c\sin\theta_0)\right)\delta(\omega-n\omega_L)
\end{align}
We use the identity
\begin{align}
    J_{n+1}(x)-J_{n-1}(x)=-2 \frac{d J_n(x)}{d x}
\end{align}
and we obtain
\begin{align}
    A_\phi({\bm x},\omega)=\sum\limits_{n=-\infty}^{\infty}A_{\phi n}({\bm x})\delta(\omega-n\omega_L)
\end{align}
and
\begin{align}
    A_{\phi n}({\bm x})=\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik_n x_0}}{x_0}\left(i\frac{\omega_L}ca\right)e^{in\phi_0}(-i)^nJ'_n(k_na\sin\theta_0)
\end{align}

The first order Fourier components of the potentials are
\begin{tcolorbox}[colback=blue!5!white,colframe=blue!75!black]
    \begin{align}
         & \Phi_n({\bm x}_0)=\frac{e_0}{4\pi\epsilon_0}\frac{e^{ik_n |{\bm x}_0|}}{|{\bm x}_0|}(-i)^ne^{in\phi_0}J_n(k_na\sin\theta_0)\nonumber                                                           \\
         & A_{rn}({\bm x}_0)= \frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik_n |{\bm x}_0|}}{|{\bm x}_0|}(-i)^ne^{in\phi_0}J_n(k_na\sin\theta_0)                                                                  \\
         & A_{\theta n}({\bm x}_0)= \frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik_n |{\bm x}_0|}}{|{\bm x}_0|}(-i)^ne^{in\phi_0}\cot\theta_0J_n(k_na\sin\theta_0)                                                \\
         & A_{\phi n}({\bm x}_0)=\frac{e_0}{4\pi\epsilon_0c}\frac{e^{ik_n |{\bm x}_0|}}{|{\bm x}_0|}(-i)^ne^{in\phi_0}(ik_La)J'_n(k_na\sin\theta_0), \quad \omega_n=n\omega_L, \quad k_n=\frac{\omega_n}c
    \end{align}
\end{tcolorbox}




\subsection{The second order expansion}


\subsubsection{The scalar potential}

The second order term from the expression pf the scalar potential is
\begin{align}
    \Phi({\bm x}_0,\omega) & =\frac{e_0}{4\pi\epsilon_0}\frac1{2\pi}\frac{e^{ikx_0}}{x_0^2}\int\limits_{-\infty}^{\infty}dt e^{i\omega t-i\frac{\omega}ca\sin\theta_0\cos(\omega_Lt-\phi_0)}\left({{\bm n}_0\cdot{\bm r}_0(t)}+\frac{i\omega}{2c}{|{\bm n}_0\times{\bm r}_0(t)|^2}       \right)\nonumber                                    \\
                           & \hspace*{-1cm}=\frac{e_0}{4\pi\epsilon_0}\frac1{2\pi}\frac{e^{ikx_0}}{x_0^2}\int\limits_{-\infty}^{\infty}dt e^{i\omega t-i\frac{\omega}ca\sin\theta_0\cos(\omega_Lt-\phi_0)}\left[a\sin\theta_0\cos(\omega_Lt-\phi_0) +\frac{ik}2a^2\left(\cos^2\theta_0+\sin^2\theta_0\sin^2(\omega_Lt -\phi_0)\right)\right]
\end{align}
We use the formula
\begin{align}
    \cos^2\theta_0+\sin^2\theta_0\sin^2(\omega_Lt-\phi_0)= & \cos^2\theta_0-\sin^2\theta_0\frac14\left(e^{2i(\omega_Lt-\phi_0)}-2+e^{-2i(\omega_Lt-\phi_0)}\right)\nonumber \\
    =                                                      & \frac12(1+\cos^2\theta_0)-\frac14\sin^2\theta_0(e^{2i(\omega_Lt -\phi_0)}+e^{-2i(\omega_Lt -\phi_0)})
\end{align}
and the expansion
\begin{align}
    e^{-ia\sin\theta_0\cos(\omega_Lt-\phi_0)}=\sum_{n=-\infty}^{\infty} (-i)^n J_n(\frac{\omega a}c\sin\theta_0) e^{-i n (\omega_Lt-\phi_0)}
\end{align}
The scalar potential becomes
\begin{align}
    \Phi({\bm x}_0,\omega) & =\frac{e_0}{4\pi\epsilon_0}\frac{e^{ikx_0}}{x_0^2}\frac1{2\pi}\sum\limits_{n=-\infty}^{\infty}(-i)^nJ_n(\frac{\omega a}c\sin\theta_0)\nonumber                     \\
                           & \times\left[\frac{a\sin\theta_0}2\int\limits_{-\infty}^{\infty}dt\left(e^{i\omega t-i(n-1)(\omega_Lt-\phi_0)}+e^{i\omega t-i(n+1)(\omega_Lt-\phi_0)}\right)\right.
    +\frac12(1+\cos^2\theta_0)\int\limits_{-\infty}^{\infty}dte^{i\omega t-in(\omega_Lt-\phi_0)}\nonumber                                                                                       \\
                           & \left.\quad-\frac14\sin^2\theta_0\int\limits_{-\infty}^{\infty}dt\left(e^{i\omega t-i(n-2)(\omega_Lt-\phi_0)}+e^{i\omega t-i(n+2)(\omega_Lt-\phi_0)}\right)\right]
\end{align}
The integrals become delta functions and the expression of the potential becomes
\begin{align}
    \Phi({\bm x}_0,\omega) & =\frac{e_0}{4\pi\epsilon_0}\frac{e^{ikx_0}}{x_0^2}\sum\limits_{n=-\infty}^{\infty}(-i)^nJ_n(\frac{\omega a}c\sin\theta_0)\nonumber                      \\
                           & \times\left[\frac{a\sin\theta_0}2\left(\delta(\omega-(n-1)\omega_L)e^{i(n-1)\phi_0}+\delta(\omega-(n+1)\omega_L)e^{i(n+1)\phi_0}\right)\right.\nonumber \\
                           & \quad+\frac{ika^2}{4}(1+\cos^2\theta_0)\delta(\omega-n\omega_L)e^{in\phi_0}\nonumber                                                                    \\
                           & \left.\quad-\frac{ika^2}8\sin^2\theta_0\left(\delta(\omega-(n-2)\omega_L)e^{i(n-2)\phi_0}+\delta(\omega-(n+2)\omega_L)e^{i(n+2)\phi_0}\right)\right]
\end{align}
We change the summation indices such that we only have $\delta(\omega-n\omega_L)$ in each term. The argument of the $\delta functions$ will be only $ak_n\sin\theta_0$; the exponential $e^{in\phi_0}$ will be a common factor.
\begin{align}
    \Phi({\bm x}_0,\omega) & =\frac{e_0}{4\pi\epsilon_0}\sum\limits_{n=-\infty}^{\infty}\frac{e^{ik_nx_0}}{x_0^2}(-i)^nJe^{in\phi_0}\delta(\omega-n\omega_L)\nonumber \\
                           & \times\left[\frac{a\sin\theta_0}2(-i)\left(J_{n+1}(k_na\sin\theta_0)-J_{n-1}(k_na\sin\theta_0)\right)\right.\nonumber                    \\
                           & \quad+\frac{ik_na^2}{2}J_n(k_na\sin\theta_0)-2\frac{ik_na^2}8\sin^2\theta_0J_n(k_na\sin\theta_0)\nonumber                                \\
                           & \left.\quad+\frac{ik_na^2}8\sin^2\theta_0\left(J_{n-2}(k_na\sin\theta_0)+J_{n+2}(k_na\sin\theta_0)\right)\right]
\end{align}
We use the identities
\begin{align}
    J_{n+1}(x)-J_{n-1}(x)=-2J_n'(x),\qquad J_n(x)-J_{n-2}(x)=-2J_{n-1}'(x),\qquad J_{n+2}(x)-J_n(x)=-2J'_{n+1}(x)
\end{align}
and write the potential as
\begin{align}
    \Phi({\bm x}_0,\omega) & =\frac{e_0}{4\pi\epsilon_0}\sum\limits_{n=-\infty}^{\infty}\frac{e^{ik_nx_0}}{x_0^2}(-i)^ne^{in\phi_0}\delta(\omega-n\omega_L)\nonumber                       \\
                           & \times\left[{ia\sin\theta_0}J'_{n}(k_na\sin\theta_0)+\frac{ik_na^2}{2}J_n(k_na\sin\theta_0)\right.\nonumber                                                   \\
                           & \quad \left.-\frac{ik_na^2}8\sin^2\theta_0\left(J_n(k_na\sin\theta_0)-J_{n-2}(k_na\sin\theta_0)-J_{n+2}(k_na\sin\theta_0)+J_n(k_na\sin\theta_0)\right)\right]
\end{align}
\begin{align}
